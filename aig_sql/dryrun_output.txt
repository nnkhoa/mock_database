--- S3: khách ACC tăng trưởng nhưng giảm biên + tăng DSO
WITH cur AS (
  SELECT customer_id, SUM(net_revenue_vnd) dt, SUM(gross_profit_vnd) gp
  FROM fact_sales WHERE entity_id='ACC' AND invoice_date BETWEEN '2026-07-01' AND '2026-09-30' GROUP BY customer_id),
prev AS (
  SELECT customer_id, SUM(net_revenue_vnd) dt, SUM(gross_profit_vnd) gp
  FROM fact_sales WHERE entity_id='ACC' AND invoice_date BETWEEN '2025-07-01' AND '2025-09-30' GROUP BY customer_id)
SELECT c.customer_id, d.customer_name, c.dt/1e9 dt_q3_26, (c.dt/p.dt-1)*100 tang_truong,
       p.gp/p.dt*100 gm_q3_25, c.gp/c.dt*100 gm_q3_26,
       (SELECT dso_days FROM fact_receivable_snapshots r WHERE r.customer_id=c.customer_id AND r.entity_id='ACC' AND snapshot_date='2026-09-30') dso_now,
       (SELECT dso_days FROM fact_receivable_snapshots r WHERE r.customer_id=c.customer_id AND r.entity_id='ACC' AND snapshot_date='2026-03-31') dso_t3
FROM cur c JOIN prev p USING(customer_id) JOIN dim_customers d USING(customer_id)
ORDER BY c.dt DESC LIMIT 10;

       SUM(i.inventory_value_vnd)/1e9 ton_ty,
       SUM(CASE WHEN i.days_to_expiry < 92 THEN i.inventory_value_vnd ELSE 0 END)/1e9 can_date_ty,
